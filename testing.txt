INSERT INTO A (WorkflowID, StepName)
SELECT NEW_WorkflowID, StepName
FROM A
WHERE WorkflowID = OLD_WorkflowID;

-- Create a temp table to store the mapping
CREATE TABLE #StepMapping (OldStepID INT, NewStepID INT);

-- Insert steps and capture the mapping
INSERT INTO A (WorkflowID, StepName)
OUTPUT INSERTED.StepID, A.StepID INTO #StepMapping
SELECT NEW_WorkflowID, StepName
FROM A
WHERE WorkflowID = OLD_WorkflowID;

INSERT INTO B (StepID, AssigneeName)
SELECT M.NewStepID, B.AssigneeName
FROM B
JOIN #StepMapping M ON B.StepID = M.OldStepID;
